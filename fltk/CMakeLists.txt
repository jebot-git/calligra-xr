find_package(ECM 6.0 REQUIRED NO_MODULE)
list(APPEND CMAKE_MODULE_PATH ${ECM_MODULE_PATH})
include(KDEInstallDirs6)
include(GNUInstallDirs)
include(GenerateExportHeader)
find_package(Qt6 6.5 REQUIRED COMPONENTS Core Xml)
find_package(KF6CoreAddons REQUIRED)
find_package(KF6Config REQUIRED)
find_package(KF6I18n REQUIRED)
find_package(Eigen3 REQUIRED NO_MODULE)
find_package(FLTK 1.4.5 EXACT CONFIG REQUIRED)
find_package(OpenGL REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(NATIVE REQUIRED IMPORTED_TARGET fontconfig zlib libzip)
set(CMAKE_AUTOMOC ON)
set(CALLIGRA_VERSION ${PROJECT_VERSION})
set(CALLIGRA_SOVERSION 41)
file(WRITE "${CMAKE_BINARY_DIR}/calligra-version.h" "#pragma once\n#define CALLIGRA_VERSION_STRING \"${PROJECT_VERSION}\"\n")
set(SHOULD_BUILD_DEVEL_HEADERS OFF)
set(SHEETS_ROOT "${CMAKE_SOURCE_DIR}/sheets")
add_subdirectory(../libs/plugin "${CMAKE_CURRENT_BINARY_DIR}/plugin")
target_include_directories(koplugin PUBLIC "${CMAKE_CURRENT_BINARY_DIR}/plugin")
add_subdirectory(../sheets/engine "${CMAKE_CURRENT_BINARY_DIR}/engine")
file(GLOB FUNCTION_SOURCES CONFIGURE_DEPENDS "${SHEETS_ROOT}/functions/*.cpp" "${SHEETS_ROOT}/functions/*Module.h")
target_sources(calligrasheetsengine PRIVATE ${FUNCTION_SOURCES})
target_compile_definitions(calligrasheetsengine PRIVATE SHEETS_NO_PLUGINMODULES)
target_include_directories(calligrasheetsengine PUBLIC
    "${SHEETS_ROOT}" "${SHEETS_ROOT}/engine" "${CMAKE_CURRENT_BINARY_DIR}/engine"
    "${CMAKE_SOURCE_DIR}/interfaces" "${CMAKE_SOURCE_DIR}/libs/flake")
target_link_libraries(calligrasheetsengine PRIVATE Eigen3::Eigen)
add_library(calligra-fltk-document STATIC src/workbook.cpp src/xlsx_import.cpp src/odf_package.cpp src/text_documents.cpp)
target_include_directories(calligra-fltk-document PUBLIC src)
target_link_libraries(calligra-fltk-document PUBLIC calligrasheetsengine PkgConfig::NATIVE)
add_library(calligra-fltk-ui STATIC src/main.cpp)
target_link_libraries(calligra-fltk-ui PUBLIC calligra-fltk-document fltk::gl-shared fltk::images-shared fltk::fltk-shared OpenGL::GL)
include(src/suite/desktop-menu.cmake)
de_xr_desktop_menu(calligra-fltk-ui)
target_compile_options(calligra-fltk-ui PRIVATE -Wall -Wextra -Wpedantic)
foreach(app words sheets stage)
    add_executable(calligra-${app}-fltk src/${app}_main.cpp)
    target_link_libraries(calligra-${app}-fltk PRIVATE calligra-fltk-ui)
    set_target_properties(calligra-${app}-fltk PROPERTIES INSTALL_RPATH "$ORIGIN/../${CMAKE_INSTALL_LIBDIR};/opt/fltk-vr/toolkit/lib")
    install(TARGETS calligra-${app}-fltk RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endforeach()
set_target_properties(calligrasheetsengine koplugin PROPERTIES INSTALL_RPATH "$ORIGIN")
install(FILES README.md PORT_STATUS.md VALIDATION.md DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/calligra-fltk)
install(FILES "${CMAKE_SOURCE_DIR}/COPYING" "${CMAKE_SOURCE_DIR}/COPYING.LIB" DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/calligra-fltk)
install(DIRECTORY "${CMAKE_SOURCE_DIR}/LICENSES/" DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/calligra-fltk/LICENSES)
install(DIRECTORY "${SHEETS_ROOT}/functions/" DESTINATION ${CMAKE_INSTALL_DATADIR}/calligrasheets/functions FILES_MATCHING PATTERN "*.xml")
include(CTest)
if(BUILD_TESTING)
    add_executable(calligra-fltk-test tests/workbook_test.cpp)
    target_link_libraries(calligra-fltk-test PRIVATE calligra-fltk-document)
    target_compile_options(calligra-fltk-test PRIVATE -UNDEBUG)
    add_test(NAME fltk-workbook COMMAND calligra-fltk-test)
endif()
if(BUILD_TESTING)
    add_executable(calligra-fltk-xlsx-test tests/xlsx_test.cpp)
    target_link_libraries(calligra-fltk-xlsx-test PRIVATE calligra-fltk-document)
    target_compile_options(calligra-fltk-xlsx-test PRIVATE -UNDEBUG)
    add_test(NAME fltk-xlsx COMMAND calligra-fltk-xlsx-test)
endif()
