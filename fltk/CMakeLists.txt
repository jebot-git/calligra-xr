include(GNUInstallDirs)
include(cmake/SystemFLTK.cmake)
find_package(OpenGL REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(NATIVE REQUIRED IMPORTED_TARGET fontconfig>=2.17 freetype2 harfbuzz zlib libzip libxml-2.0 glib-2.0 icu-uc icu-i18n)
add_library(calligra-fltk-document STATIC src/workbook.cpp src/xlsx_import.cpp src/ooxml_text_import.cpp src/odf_package.cpp src/text_documents.cpp src/native_words_edit.cpp src/native_words_tables.cpp src/native_words_format.cpp src/native_words_page.cpp src/native_formula.cpp src/native_sheets.cpp src/native_engineering.cpp src/native_financial.cpp src/native_financial_dates.cpp src/native_calendar.cpp src/native_coupon.cpp src/native_bond.cpp src/native_security.cpp src/native_amortization.cpp src/native_euro.cpp src/native_text_layout.cpp src/native_table_model.cpp src/native_table_layout.cpp src/native_table_pagination.cpp src/native_text_cursor.cpp src/native_text_keep.cpp src/native_table_stream.cpp src/native_text_shaper.cpp src/native_text_hit.cpp src/native_text_editing.cpp src/native_list_number.cpp src/native_tabs.cpp src/native_image.cpp src/native_presentation.cpp src/native_path.cpp src/native_enhanced_formula.cpp src/native_enhanced_geometry.cpp)
target_include_directories(calligra-fltk-document PUBLIC src)
target_link_libraries(calligra-fltk-document PUBLIC PkgConfig::NATIVE)
pkg_check_modules(PRINT REQUIRED IMPORTED_TARGET cairo-pdf cairo-ft gio-2.0)
add_library(calligra-fltk-print STATIC src/native_print.cpp src/native_image_decode.cpp)
target_link_libraries(calligra-fltk-print PUBLIC calligra-fltk-document PkgConfig::PRINT fltk::images-shared fltk::fltk-shared)
add_library(calligra-fltk-ui STATIC src/main.cpp src/print_dialog.cpp src/format_dialog.cpp src/text_page_canvas.cpp src/gpu_text.cpp src/gpu_image.cpp src/slide_canvas.cpp)
target_link_libraries(calligra-fltk-ui PUBLIC calligra-fltk-print calligra-fltk-document fltk::gl-shared fltk::images-shared fltk::fltk-shared OpenGL::GL OpenGL::GLU)
include(src/suite/desktop-menu.cmake)
de_xr_desktop_menu(calligra-fltk-ui)
target_compile_options(calligra-fltk-ui PRIVATE -Wall -Wextra -Wpedantic)
foreach(app words sheets stage)
    add_executable(calligra-${app}-fltk src/${app}_main.cpp)
    target_link_libraries(calligra-${app}-fltk PRIVATE calligra-fltk-ui)
    set_target_properties(calligra-${app}-fltk PROPERTIES INSTALL_RPATH "$ORIGIN/../${CMAKE_INSTALL_LIBDIR}")
    install(TARGETS calligra-${app}-fltk RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endforeach()

install(FILES README.md PORT_STATUS.md VALIDATION.md DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/calligra-fltk)
install(FILES "${CMAKE_SOURCE_DIR}/COPYING" "${CMAKE_SOURCE_DIR}/COPYING.LIB" DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/calligra-fltk)
install(DIRECTORY "${CMAKE_SOURCE_DIR}/LICENSES/" DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/calligra-fltk/LICENSES)
