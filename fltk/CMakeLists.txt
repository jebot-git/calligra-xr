include(GNUInstallDirs)
include(cmake/SystemFLTK.cmake)
find_package(OpenGL REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(NATIVE REQUIRED IMPORTED_TARGET fontconfig>=2.17 freetype2 harfbuzz zlib libzip libxml-2.0 glib-2.0 icu-uc icu-i18n)
add_library(calligra-fltk-document STATIC src/workbook.cpp src/xlsx_import.cpp src/ooxml_text_import.cpp src/odf_package.cpp src/text_documents.cpp src/native_formula.cpp src/native_sheets.cpp src/native_text_layout.cpp src/native_text_shaper.cpp src/native_list_number.cpp src/native_image.cpp src/native_presentation.cpp src/native_path.cpp src/native_enhanced_formula.cpp src/native_enhanced_geometry.cpp)
target_include_directories(calligra-fltk-document PUBLIC src)
target_link_libraries(calligra-fltk-document PUBLIC PkgConfig::NATIVE)
add_library(calligra-fltk-ui STATIC src/main.cpp src/text_page_canvas.cpp src/gpu_text.cpp src/gpu_image.cpp src/slide_canvas.cpp)
target_link_libraries(calligra-fltk-ui PUBLIC calligra-fltk-document fltk::gl-shared fltk::images-shared fltk::fltk-shared OpenGL::GL OpenGL::GLU)
include(src/suite/desktop-menu.cmake)
de_xr_desktop_menu(calligra-fltk-ui)
target_compile_options(calligra-fltk-ui PRIVATE -Wall -Wextra -Wpedantic)
foreach(app words sheets stage)
    add_executable(calligra-${app}-fltk src/${app}_main.cpp)
    target_link_libraries(calligra-${app}-fltk PRIVATE calligra-fltk-ui)
    set_target_properties(calligra-${app}-fltk PROPERTIES INSTALL_RPATH "$ORIGIN/../${CMAKE_INSTALL_LIBDIR}")
    install(TARGETS calligra-${app}-fltk RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endforeach()

install(FILES README.md PORT_STATUS.md VALIDATION.md DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/calligra-fltk)
install(FILES "${CMAKE_SOURCE_DIR}/COPYING" "${CMAKE_SOURCE_DIR}/COPYING.LIB" DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/calligra-fltk)
install(DIRECTORY "${CMAKE_SOURCE_DIR}/LICENSES/" DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/calligra-fltk/LICENSES)
